% COLUMN 1
\begin{tcolorbox}[colback=blue!10,colframe=blue!70!black,title=\textbf{Quick Start}]
\textbf{1. Start System}
\begin{itemize}[noitemsep,leftmargin=*]
    \item Open Fusion 360
    \item SOLID → CREATE → Run Order
    \item Click YES (Start Monitor)
\end{itemize}

\textbf{2. Submit Order}
\begin{itemize}[noitemsep,leftmargin=*]
    \item Create JSON file
    \item Copy to \texttt{order\_dropbox}
    \item Wait for completion
\end{itemize}

\textbf{3. Find Outputs}
\begin{itemize}[noitemsep,leftmargin=*]
    \item Check \texttt{output} folder
    \item Look for order ID folder
    \item Retrieve files
\end{itemize}
\end{tcolorbox}

\vspace{0.2cm}

\begin{tcolorbox}[colback=green!10,colframe=green!70!black,title=\textbf{Folder Structure}]

\begin{tcolorbox}[colback=red!10,colframe=red!70!black,title=\textbf{Troubleshooting}]
\textbf{Order Failed?}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item Check \texttt{order\_failed} folder
    \item Read error log (.txt file)
    \item Fix issue in JSON
    \item Re-submit order
\end{enumerate}

\textbf{No Output Files?}
\begin{itemize}[noitemsep,leftmargin=*]
    \item Verify monitor is running
    \item Check \texttt{order\_processing}
    \item Check log file in \texttt{output\textbackslash logs}
    \item Verify network access
\end{itemize}

\textbf{Restart System:}
\begin{enumerate}[noitemsep,leftmargin=*]
    \item Utilities → ADD-INS
    \item Stop add-in
    \item Run add-in
    \item Start monitor
\end{enumerate}
\end{tcolorbox}
